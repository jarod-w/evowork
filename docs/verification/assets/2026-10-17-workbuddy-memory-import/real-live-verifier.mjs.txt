 const r=await page.evaluate(input=>window.evowork.send(input),{workspaceId:ids[project],modelId,text:marker+'\n'+prompt,...(useMemories===undefined?{}:{useMemories})});
 let history;const end=Date.now()+180000;

 const task=(await page.evaluate(()=>window.evowork.getStartup())).tasks.find(t=>t.id===r.threadId);
 const answer=history.items.filter(i=>i.type==='agentMessage'&&i.phase!=='commentary').map(i=>i.text).join('\n');
 const record={name,project,useMemories,task,status:history.turns?.at(-1)?.status,answer,items:history.items,wire:wire().filter(r=>r.cases.includes(marker))};result.cases.push(record);
 check(record.status==='completed',name+' completed');check(task.cwd===seed.paths[project],name+' cwd');check(record.wire.length>0,name+' observed');save();
